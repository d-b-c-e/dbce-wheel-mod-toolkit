cmake_minimum_required(VERSION 3.20)
project(dbce_support_report LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Native support file IO currently targets Windows x86/x64 hosts")
endif()
add_library(dbce_support_report STATIC support_report.cpp)
target_compile_features(dbce_support_report PUBLIC cxx_std_17)
target_include_directories(dbce_support_report PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
include(CTest)
if(BUILD_TESTING)
    add_executable(dbce_support_report_tests test_support_report.cpp)
    target_link_libraries(dbce_support_report_tests PRIVATE dbce_support_report)
    add_test(NAME support_report COMMAND dbce_support_report_tests)
    set_tests_properties(support_report PROPERTIES TIMEOUT 30)
endif()
if(MSVC)
    target_compile_options(dbce_support_report PRIVATE /W4 /WX /utf-8 /permissive-)
    if(BUILD_TESTING)
        target_compile_options(dbce_support_report_tests PRIVATE /W4 /WX /utf-8 /permissive-)
    endif()
endif()
